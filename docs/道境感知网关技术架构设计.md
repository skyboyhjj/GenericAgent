# 道境感知网关技术架构设计

## 一、架构总览

道境感知网关是慧惠数字伴侣的“感官系统”，负责从多种感知源（小程序、可穿戴设备、VR/AR等）统一采集数据，标准化为“道境感知流”，供蒸馏管道和记忆系统消费。

### 设计原则

- **可插拔**：每种感知源实现统一接口，新增感知源无需改动核心
- **静默无感**：所有数据同步在后台异步执行，不阻塞用户交互
- **数据主权**：所有数据保留在本地，仅通过127.0.0.1传输
- **为道日损**：网关负责“无损”采集原始数据，蒸馏由管道层负责

### 3.1 感知源适配层 (Adapter Layer)

#### 抽象感知源接口

```python
from abc import ABC, abstractmethod
from enum import Enum
from typing import List, Optional
from dataclasses import dataclass, field
from datetime import datetime

class SyncMode(Enum):
    FULL = "full"          # 全量同步（首次连接）
    INCREMENTAL = "incremental"  # 增量同步（日常更新）
    RESYNC = "resync"      # 重新同步（用户手动触发）

class SourceType(Enum):
    DAO_MING = "dao_ming"      # 袭明每日镜鉴
    WEARABLE = "wearable"       # 可穿戴设备
    VR_AR = "vr_ar"            # VR/AR眼镜
    IOT = "iot"                # 物联网传感器
    BCI = "bci"                # 脑机接口（远期）

@dataclass
class ConnectivityStatus:
    """连通性状态"""
    reachable: bool
    latency_ms: float
    error_message: str = ""

@dataclass
class SyncResult:
    """同步结果"""
    success: bool
    source_type: str
    data: List[dict] = field(default_factory=list)
    error_message: str = ""
    synced_at: str = ""
    data_date: str = ""
    algorithm_version: str = ""

class PerceptionSource(ABC):
    """所有感知源必须实现的统一接口"""
    
    @abstractmethod
    async def check_connectivity(self) -> ConnectivityStatus:
        """检测感知源是否可达"""
        ...
    
    @abstractmethod
    async def sync(self, mode: SyncMode) -> SyncResult:
        """同步数据"""
        ...
    
    @abstractmethod
    def get_source_type(self) -> SourceType:
        """返回感知源类型"""
        ...
    
    @abstractmethod
    def get_data_schema(self) -> dict:
        """返回感知源的数据模式描述"""
        ...
```

### 3.2 数据融合层 (Fusion Layer)

负责将多个感知源的异构数据，按时间对齐，生成统一的“道境感知流”。

```python
@dataclass
class MultiModalEvent:
    """多模态统一事件"""
    timestamp: datetime
    source_type: SourceType
    event_type: str          # "修行记录" / "生理数据" / "环境数据"
    data: dict               # 原始数据
    normalized: dict         # 标准化后的通用字段

class MultiModalEventStream:
    """多模态事件流"""
    
    def __init__(self):
        self.events: List[MultiModalEvent] = []
    
    def add_events(self, events: List[MultiModalEvent]):
        self.events.extend(events)
        self._sort_by_timestamp()
    
    def get_by_source(self, source_type: SourceType) -> List[MultiModalEvent]:
        return [e for e in self.events if e.source_type == source_type]
    
    def get_by_timerange(self, start: datetime, end: datetime) -> List[MultiModalEvent]:
        return [e for e in self.events if start <= e.timestamp <= end]
    
    def _sort_by_timestamp(self):
        self.events.sort(key=lambda e: e.timestamp)


class DataFusionEngine:
    """数据融合引擎"""
    
    def __init__(self):
        self.time_aligner = TimeAlignmentEngine()
        self.event_correlator = EventCorrelationEngine()
        self.context_wrapper = ContextWrapper()
    
    async def fuse(self, sync_results: List[SyncResult]) -> MultiModalEventStream:
        """将多个感知源的同步结果融合为统一事件流"""
        
        # 1. 将各感知源的原始数据，统一包装为 MultiModalEvent
        raw_events = []
        for result in sync_results:
            if result.success:
                mapper = self._get_event_mapper(result.source_type)
                raw_events.extend(mapper.map(result.data))
        
        # 2. 时间对齐
        aligned_events = self.time_aligner.align(raw_events)
        
        # 3. 事件关联
        correlated_events = self.event_correlator.correlate(aligned_events)
        
        # 4. 上下文包装
        wrapped_stream = self.context_wrapper.wrap(correlated_events)
        
        return wrapped_stream
```

### 3.3 调度层 (Orchestration Layer)

管理同步的生命周期、策略与健康降级。

```python
class SyncPolicy(Enum):
    ON_SIGNAL = "on_signal"        # 信号触发（如晚间回顾完成）
    SCHEDULED = "scheduled"         # 定时调度（如每30分钟）
    ON_DEMAND = "on_demand"         # 用户主动触发
    LAZY = "lazy"                  # 用户对话时懒加载

class DaoGateway:
    """道境感知网关——总调度器"""
    
    def __init__(self):
        self.sources: Dict[SourceType, PerceptionSource] = {}
        self.fusion_engine = DataFusionEngine()
        self.policy_manager = SyncPolicyManager()
        self.health_monitor = HealthMonitor()
    
    def register_source(self, source: PerceptionSource):
        """注册一个新的感知源"""
        self.sources[source.get_source_type()] = source
        # 启动健康监控
        asyncio.create_task(self.health_monitor.monitor(source))
    
    async def sync_all(self, mode: SyncMode = SyncMode.INCREMENTAL):
        """并发同步所有已注册感知源"""
        tasks = []
        for source_type, source in self.sources.items():
            tasks.append(self._sync_with_fallback(source, mode))
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 过滤异常，保留有效结果
        valid_results = [r for r in results if isinstance(r, SyncResult)]
        
        # 融合为多模态事件流
        stream = await self.fusion_engine.fuse(valid_results)
        
        # 输出到蒸馏管道
        await self._pipe_to_pipeline(stream)
        
        return stream
    
    async def _sync_with_fallback(self, source: PerceptionSource, mode: SyncMode) -> SyncResult:
        """带降级策略的同步"""
        connectivity = await source.check_connectivity()
        
        if not connectivity.reachable:
            return SyncResult(
                success=False,
                source_type=source.get_source_type().value,
                error_message=f"不可达: {connectivity.error_message}"
            )
        
        return await source.sync(mode)
    
    async def _pipe_to_pipeline(self, stream: MultiModalEventStream):
        """将融合后的事件流输出到蒸馏管道"""
        from memory.ximing_pipeline import XiMingPipeline
        pipeline = XiMingPipeline()
        await pipeline.process_multimodal_stream(stream)
```

## 五、与 S2-1 的关系

当前 Sprint 2 的 S2-1 任务（实现 XiMingClient）将直接产出 `XiMingAdapter`。本架构设计是在此基础上的**战略升维**——将单一适配器重构为统一网关的可插拔组件。

**实施路径**：
1. **S2-1 当前**：实现 `XiMingAdapter` 作为独立的 `memory/ximing_client.py`
2. **S2-1 之后**：创建 `memory/gateway/` 目录结构，将 `XiMingAdapter` 迁移为 `memory/gateway/adapters/ximing_adapter.py`，并实现 `DaoGateway` 核心框架
3. **Sprint 3+**：陆续实现其他感知源适配器，完善融合引擎

这样我们既不打乱当前 Sprint 的节奏，又为慧惠的“归真”与“合道”之路铺设了坚实的工程基础。
